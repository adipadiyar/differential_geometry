\begin{definition}
  A submanifold $S$ is called totally geodesic if for every $p \in S$ and $v \in T_pS$ we have $\text{range } \gamma_v \subseteq S$
\end{definition}

\begin{definition}
  We say that $X \in \mathcal{X}(M)$ is compatible with $S$ if $X_{|S} \in \mathcal{X}(S)$. This space is $\mathcal{X}_S(M)$
\end{definition}

\begin{claim}
  If $X,Y \in \mathcal{X}_S(M)$ then $[X,Y]_{|S}=[X_{|S},Y_{|S}]_S$
\end{claim}

\section{Second fundamental form}

Fix a submanifold $S$. Let $\nabla$ and $\tilde{\nabla}$ denote the Levi-Civita connections on $M$ and $S$ respectively.

\begin{definition}
  Define $\mathcal{F} \in B(\mathcal{X}_{S}(M),C^\infty(S,\mathbb{R}^k))$ by \[X,Y  \to (\nabla_XY)-\tilde{\nabla}_XY\]
\end{definition}

\begin{claim}
  $\text{range }\mathcal{F} \subseteq \mathcal{Y}(S)$
\end{claim}

\begin{proof}
  Observe that $[\tilde{\nabla}_XY](q)=\tilde{\pi}([\nabla_XY](q))$
\end{proof}

\begin{claim}
  $\mathcal{F}$ is symmetric
\end{claim}

\begin{claim}
  $\mathcal{F}(fX,Y)=f\mathcal{F}(X,Y)$ for any $f \in C^\infty(M)$
\end{claim}

\begin{claim}
  If $X(p)=0$ then $\mathcal{F}(X,Y)(p)=0$
\end{claim}

\begin{proof}
  Choose a tangential frame $X_i \in \mathcal{X}(U)$ on some neighborhood $U$. We have $X=\sum f_iX_i$ with $f_i(p)=0$. Now use the bilinearity with respect to smooth functions.
\end{proof}

\begin{claim}
  Suppose $v \in T_pS$. Then there is a neighborhood $U$ of $p$ and $X \in \mathcal{X}_{S \cap U}(U)$ such that $X(p)=v$
\end{claim}

\begin{proof} 
  Let $\phi:\Omega \to U$ be a local parametrization compatible with $S$, with inverse $\psi$. 
  Let $\varphi:W \to S \cap U$ is the associated local parametrization of $S$. 
  Now $v=(D\varphi)_0y$. Define $X: U \to \mathbb{R}^k$ by $X(q)=(D\psi)_q^{-1}(y,0)$
\end{proof}

\begin{claim}
  Suppose $v_1,v_2 \in T_pS$ and $X_i,Y_i \in \mathcal{X}_S(M)$ are such that \[X_i(p)=Y_i(p)=v_i\]
  Then $\mathcal{F}(X_1,X_2)(p)=\mathcal{F}(Y_1,Y_2)(p)$
\end{claim}

\begin{proof}
  We have $\mathcal{F}(X_1,X_2)-\mathcal{F}(Y_1,Y_2)=\mathcal{F}(X_1-Y_1,X_2)-\mathcal{F}(Y_1,X_2-Y_2)$. But $X_1-Y_1$ and $X_2-Y_2$ vanish at $p$
\end{proof}

\begin{definition}
  We have $Q_p \in B(T_pS,N_pS)$ defined by $(v,w) \to \mathcal{F}(X,Y)(p)$
\end{definition}

\begin{claim}
  $Q_p$ is symmetric
\end{claim}

\begin{claim}\label{relatingCov}
  Suppose $\gamma$ is a smooth curve in $S$ defined on an interval $I$ and $V$ is a smooth vector field along $\gamma$ satisfying $V(t) \in T_{\gamma(t)}S$.
  Then we have \[(\text{cov }V)(t)=(\widetilde{\text{cov }}V)(t)+Q_{\gamma(t)}(\gamma'(t),V(t))\]
  where $\text{cov}$ and $\widetilde{\text{cov}}$ denote the covariant derivatives along $\gamma$ in $M$ and $S$ respectively. 
\end{claim}

\begin{proof}
  Consider the following
  \begin{itemize}
    \item Choose $t_0$ and a compatible local parametrization $\phi:\Omega \to U$
    \item The associated local parametrization of $S$ is $\varphi:W \to S \cap U$
    \item Let $\gamma(t_0)=p$ and $T:\mathbb{R}^k \to N_pS$ be the orthogonal projection
    \item $V(t)=V_i(t)\varphi_i(\gamma(t))$ and $\gamma(t)=\varphi(x(t))$
    \item $Q_p(\gamma'(t_0),V(t_0)) = \dot{x_i}(t_0)V_j(t_0)Q_p(\varphi_i(p),\varphi_j(p))$
    \item $\dot{V}(t_0)=\dot{x_i}(t_0)V_j(t_0)(D_{\varphi_i}\varphi_j)(p)+\dot{V_j}(t_0)\varphi_j(p)$
    \item $(\text{cov}-\widetilde{\text{cov}})V(t_0)=T(\dot{V}(t_0))=\dot{x_i}(t_0)V_j(t_0)T(D_{\varphi_i}\varphi_j)(p)$ 
    \item Now use the definition of $Q_p$. The required vector field extensions are $\phi_i$ and $\phi_j$ respectively
  \end{itemize}
\end{proof}

\begin{claim}\label{nullQ}
  If $S$ is totally geodesic then $Q_p=0$ for all $p \in S$
\end{claim}

\begin{proof}
  Let $v \in T_pS$. 
  Applying the previous claim to $\gamma_v$ we get \[\widetilde{\text{cov}}\gamma'(0)+Q_p(v,v)=0\]
  Since the first term is in $T_pS$ and the second term is in $N_pS$ we get that $Q_p(v,v)=0$.
  Since $Q_p$ is symmetric it must be zero.
\end{proof}

\section{Fixed points}

Suppose $\mathcal{A} \subseteq \text{Iso}(M)$, where $\text{Iso}(M)$ denotes the group of Riemannian isometries on $M$.
Define \[C=\{q \in M: q \text{ is a fixed point of every }A \in \mathcal{A}\}\]
For $q \in C$ define the subspace $W_q$ of $T_qM$ to be the points which are fixed by each $DA_q$, for $A \in \mathcal{A}$ 

\begin{claim}
  $C$ is closed in $M$
\end{claim}

\begin{claim}\label{WtoC}
  Suppose $q \in C$ and $v \in \mathcal{E}_q \cap W_q$. Then $\exp_q(v) \in C$
\end{claim}

\begin{proof}
  Let $A \in \mathcal{A}$. We have $A\exp_q(v)=\exp_q(DA_qv)=\exp_q(v)$, where the last equality holds because $v \in W_q$.
\end{proof}

\begin{claim}\label{CWrelation}
  Let $q \in C$ and $U=\exp_q(B_\rho)$, where $\rho$ is the injectivity radius. Then $C \cap U = \exp_q(W_q \cap B_\rho)$
\end{claim}

\begin{proof}
  Consider the following:
  \begin{itemize}
    \item RHS is contained in LHS by \ref{WtoC}
    \item If $\exp_q(v) \in C$ with $||v||<\rho$, then $\exp_q(DA_qv)=A\exp_q(v)=\exp_q(v)$. Since $DA_q$ is an isometry, $v \in W_q$
  \end{itemize}
\end{proof}

\begin{claim}
  $C$ is a submanifold of $M$
\end{claim}

\begin{proof}
  We have that $W_q \cap B_\rho$ is a submanifold of $B_\rho$.
  Since $\exp_q:B_\rho \to U$ is a diffeomorphism, by \ref{CWrelation} $C \cap U$ is a submanifold. 
  Since this holds for all $q \in C$ and being a manifold is a local property, $C$ is a manifold.
\end{proof}

\begin{claim}
  Every connected component of a manifold is a submanifold
\end{claim}

\begin{proof}
  Every manifold is locally path connected. Now use Theorem 25.4 of Munkres.
\end{proof}

\begin{claim}\label{CtangentSpace}
  We have $T_qC=W_q$
\end{claim}

\begin{proof}
  This follows directly from \ref{CWrelation}
\end{proof}

\begin{claim}\label{totalC}
  $C$ is totally geodesic
\end{claim}

\begin{proof}
  Suppose $\gamma_v$ is the maximal geodesic at $q \in C$ with $v \in T_qC$. Now $\gamma_v(t)=\exp_q(tv) \in C$, by using \ref{CtangentSpace} and \ref{WtoC}
\end{proof}

Suppose now that $C$ is a connected one dimensional submanifold of $M$ and $\sigma: \mathbb{R} \to C$ is a diffeomorphism

\begin{claim}
  We have $(\text{cov }\sigma')(t) \in W_{\sigma(t)}$ where the covariant derivative is in $M$
\end{claim}

\begin{proof}
  By \ref{relatingCov}, as well as \ref{nullQ} and \ref{totalC}, the covariant derivative is in $C$. Now use \ref{CtangentSpace}.
\end{proof}

\begin{claim}
  If $|\sigma'(t)|=1$ then $\sigma$ is a geodesic in $M$
\end{claim}

\begin{proof}
  Consider the following:
  \begin{itemize}
    \item Differentiating, we get $\langle \sigma''(t), \sigma'(t) \rangle=0$ 
    \item Hence $\langle (\text{cov}\sigma')(t), \sigma'(t) \rangle=0$
    \item But $(\text{cov}\sigma')(t) = f(t)\sigma'(t)$
    \item Hence $f=0$
  \end{itemize}

\end{proof}

\section{Uniformly normal neighborhood}

Suppose we have that

\begin{itemize}
  \item $W$ is a uniformly $\delta$-normal neighborhood of $p$
  \item $q_0=\exp_p(v_0) \in W$
  \item $\rho$ is the injectivity radius at $p$
  \item $0<||v_0||<\rho$
  \item Define $f:W \to [0,\infty)$ by $f(q)= [d_M(q_0,q)]^2$
  \item Define $\Psi$ to be the inverse of the diffeomorphism \[\exp_{q_0}:B_\delta \to \exp_{q_0}(B_\delta)\]
\end{itemize}

\begin{claim}
  $f$ is smooth 
\end{claim}

\begin{proof}
  We have $f(q)=||\Psi(q)||^2$
\end{proof}

\begin{claim}
  $\text{grad}_f(p)=-2v_0$
\end{claim}

\begin{proof}
  Consider the following:
  \begin{itemize}
    \item Let $\gamma:[0,1] \to M$ be the minimizing geodesic from $p$ to $q_0$
    \item We have $\gamma(t)=\exp_p(tv_0)=\exp_{q_0}((1-t)w_0)$ where $\Psi(p)=w_0$
    \item By Gauss lemma, $\text{grad}_f(p)=2(d\exp_{q_0})_{w_0}(w_0)=-2\gamma'(0)=-2v_0$ 
  \end{itemize}
\end{proof}

\section{Removal of a point}

Suppose $M$ is connected and $p \in M$. Define $N = M \setminus \{p\}$

\begin{claim}
  $N$ is a connected submanifold
\end{claim}

\begin{proof}
  Consider the following:
  \begin{itemize}
    \item Since $N$ is open in $M$ it is a submanifold.
    \item Suppose $N=U_1 \sqcup U_2$ is a separation. Taking closures we get $M=\overline{U_1} \cup \overline{U_2}$.
          Since $M$ is connected these sets cannot be disjoint. Hence $p \in \overline{U_1} \cap \overline{U_2}$.
          There is a connected neighborhood $V$ of $p$ such that $V \cap N$ is also connected. 
          Without loss of generality we can assume $V$ is disjoint from $U_1$, which is a contradiction.
  \end{itemize}
\end{proof}

\begin{claim}
  For all $q_1,q_2 \in N$ we have $d_N(q_1,q_2)=d_M(q_1,q_2)$
\end{claim}

\begin{proof}
  Consider the following:
  \begin{itemize}
    \item Suppose $\gamma:[0,1] \to M$ is a curve from $q_1$ to $q_2$.
          Let $S_r$ be the geodesic sphere of radius $r$. 
          Let $t_0(r)$ and $t_1(r)$ be the first and last times at which $\gamma$ is in $S_r$.
    \item Then \[d_N(q_1,q_2) \leq d_N(q_1,\gamma(t_0(r))) + d_N(\gamma(t_0(r)),\gamma(t_1(r))) + d_N(\gamma(t_1(r)),q_2) \] using the triangle inquality
    \item We can bound this by \[\text{length}(\gamma_{|[0,t_0(r)]}) + \text{diam}(S_r) + \text{length}(\gamma_{|[t_1(r),1]}) \leq \text{length}(\gamma)+ \text{diam}(S_r)\]
    \item Now take the limit as $r \to 0$
  \end{itemize}
\end{proof}
