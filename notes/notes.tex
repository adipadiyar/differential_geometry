\section{Preliminaries}

\begin{definition}
  $N \subseteq M$ is a \textbf{submanifold} of $M$ if $N$ is a manifold.
\end{definition}

\begin{definition}
  A local parametrization $\phi:\Omega \to U$ of $M$, with inverse $\psi$ is called compatible with $S$ if $\psi(S \cap U)=\Omega \cap (\mathbb{R}^m \times \{0\})$
\end{definition}

\begin{claim}
  There is a local parametrization compatible with $S$
\end{claim}

\begin{claim}
  Suppose $\phi$ is compatible with $S$. define $W=\{x:(x,0) \in \Omega\}$. then $\varphi:W \to S \cap U$ is a local parametrization of $S$ defined by $x \to \phi(x,0)$
\end{claim}

\begin{definition}
    The \textbf{tangent bundle} $TM \subseteq \mathbb{R}^{2k}$ of $M$ is defined as $\{(p,v): p \in M,v \in T_pM\}$
\end{definition}

\begin{claim}
    $TM$ is a manifold of dimension $2n$
\end{claim}

\begin{definition}
  The unit tangent bundle $UTM \subset TM$ is defined as $\{(q,v):||v||=1\}$
\end{definition}

\begin{claim}
  $UTM$ is a submanifold of $TM$
\end{claim}

\begin{definition}
  Suppose $\phi \in \text{Diffeo}(\Omega,U)$ is a local parametrization around $p$
  \begin{itemize}
    \item We define the radial map $r \in C^\infty(U)$ by $r(\phi(x))=||x||$
    \item If $\phi(0)=p$ then we define $\phi_r: U \setminus \{p\} \to \mathbb{R}^k$ by $\phi(a) \to (D\phi)_a \left (\frac{a}{||a||} \right )$ 
  \end{itemize}
\end{definition}

\section{Notions associated with the derivative}

\begin{definition}
    Define $grad \in L(C^\infty(M),\mathcal{X}(M))$ by $\langle v,\text{grad}_f(p) \rangle = Df_p(v)$ for all $v \in T_pM$
\end{definition}

\begin{definition}
    Define $Hess_p \in L(C^\infty(M),T_pM^{(2)})$ by \[f \to ((v,w) \to \langle \nabla_v\text{grad}_f,w\rangle)\]
\end{definition}

\begin{claim}
    $\text{range }\text{Hess}_p \subseteq (T_pM)^{(2)}_{\text{sym}}$
\end{claim}

\begin{claim}
    The map $X,Y \to D_XY$ is bilinear and has the following properties:
    \begin{itemize}
        \item $D_{fX}Y=f(D_XY)$
        \item $D_X(fY)=f(D_XY) + (D_Xf)Y$
        \item $D_XY-D_YX=[X,Y]$
        \item $D_X\langle Y, Z \rangle = \langle D_XY,Z \rangle + \langle Y, D_XZ \rangle$
    \end{itemize}
\end{claim}

\begin{claim}
    Suppose $\phi: \Omega \to V$ is a local parametrization. Then $D_{\phi_i}\phi_j=\frac{\partial \phi}{\partial x_i \partial x_j}$
\end{claim}

\section{Pullback by a diffeomorphism}

Suppose $F:M \to N$ is a diffeomorphism

\begin{claim}
     Suppose $v \in T_pM$. Then for all $f \in C^\infty(M)$ we have $D_v(f \circ F)=(D_{F_*v})f$
\end{claim}

\begin{claim}
    We have $F_*[X,Y]=[F_*X,F_*Y]$
\end{claim}

\begin{claim}
    We have $\langle F_*X , F_*Y \rangle = \langle X , Y \rangle $
\end{claim}

\section{Riemannian metric}

\begin{definition}
    A diffeomorphism $F:M \to N$ is called a \textbf{Riemannian isometry} if at each point the derivative $DF_p$ is an isometry from $T_pM$ to $T_pN$
\end{definition}

\begin{definition}
    A local diffeomorphism  $F: M \to N$ at $p$ is called a \textbf{local Riemannian isometry} at $p$ if there is a neighborhood $U$ such that $F(U)$ is open and $F_{|U}:U \to F(U)$ is a Riemannian isometry
\end{definition}

\begin{definition}
    Suppose $\gamma:[a,b] \to M$ is a piecewise smooth curve. \\ A \textbf{reparametrization} of $\gamma$ is given by $\gamma \circ \phi$, where $\phi:[c,d] \to [a,b]$ is a homeomorphism with a partition $c_0,c_1,\dots,c_k$ such that $\phi_{|[c_{i-1},c_i]}$ is a diffeomorphism onto its image
\end{definition}

\begin{claim}
    For any $\gamma$ piecewise regular, the map \[c \to \text{length}(\gamma_{[a,c]})\] is a valid reparametrization, called the arc length parametrization
\end{claim}

Suppose $M$ is connected

\begin{claim}
    There is a piecewise regular curve between any two points
\end{claim}

\begin{definition}
    Define the \textbf{Riemannian metric} by \[d_M(p,q)=\text{inf}\{L(\gamma):\gamma \text{ is a piecewise smooth curve from }p \text{ to }q\}\]
\end{definition}

\begin{claim}
    The topology induced by the Riemannian metric is the same as the subspace topology $M$ inherits from $\mathbb{R}^k$
\end{claim}

\begin{definition}
    A piecewise smooth curve $\gamma$ from $p$ to $q$ is \textbf{length minimizing} if it has length $d_M(p,q)$
\end{definition}

\begin{claim}
    If there is a length minimizing curve from $p$ to $q$ then there is such a length minimizing piecewise regular curve as well 
\end{claim}

\begin{claim}
    If $M$ is closed in $\mathbb{R}^k$ then there is a length minimizing curve between any two points
\end{claim}

\begin{claim}
    Every Riemannian isometry is an isometry with respect to the Riemannian metric
\end{claim}

\begin{claim}
    Suppose $F_1,F_2$ are local Riemannian isometries with $F_1(p)=F_2(p)$ and $(DF_1)_p=(DF_2)_p$. Then $F_1=F_2$ 
\end{claim}

\section{First fundamental form}

\begin{definition}
    The \textbf{first fundamental form} of $M$ at $p$ is defined by restricting the standard inner product on $\mathbb{R}^k$ to the tangent space. We have $g_p: T_pM \times T_pM \to \mathbb{R}$
\end{definition}

\begin{definition}
For a local parametrization $\phi:\Omega \to V$ define \[G_ij(q)=g_q(\phi_i(q),\phi_j(q))\]
\end{definition}

\begin{claim}
    Two manifolds $M$ and $N$ admit a local Riemannian isometry from $p$ to $q$ if and only if there are local parametrizations $\phi: \Omega \to U$ and $\varphi : \Omega \to V$ such that $G_{ij}(\phi(a))=H_{ij}(\varphi(a))$ for all $a \in \Omega$
\end{claim}

\section{Levi-Civita connection}

\begin{definition}
    We have $\pi_p \in \text{End}(\mathbb{R}^k)$ which is the orthogonal projection onto $T_pM$. We define $\hat{\pi}_p$ as $I-\pi_p$
\end{definition}

\begin{definition}
    Suppose $v \in T_pM$. Define $\nabla_v \in L(\mathcal{X}(M),T_pM)$ by $X \to \pi_p (DX)_p v$
\end{definition}

\begin{definition}
    Suppose $X,Y \in \mathcal{X}(M)$ are smooth vector fields. Then we have the smooth vector field $\nabla_XY=(p \to \pi_p (D_XY)(p))$ called the \textbf{Levi-Civita connection} of $X$ and $Y$
\end{definition}

\begin{claim}
    The map $X,Y \to \nabla_XY$ is bilinear and has the following properties:
    \begin{itemize}
        \item $\nabla_XY-\nabla_YX = [X,Y]$
        \item $\nabla_{fX}Y=f(\nabla_XY)$
        \item $\nabla_X(fY)=f(\nabla_XY) + (D_Xf)Y$
    \end{itemize}
\end{claim}

\begin{definition}
    Suppose $\phi: \Omega \to V$ is a local parametrization. The \textbf{Christoffel symbols} $\Gamma_{ij}^k \in C^\infty(\Omega)$ are defined by $(\nabla_{\phi_i}\phi_j)(\phi(a)) = \sum_k \Gamma_{ij}^k(a) \phi_k(a)$
\end{definition}

\begin{definition}
    The \textbf{curvature tensor} of $X,Y,Z$ is defined by $R(X,Y,Z)=\nabla_X(\nabla_YZ)-\nabla_Y(\nabla_XZ)-\nabla_{[X,Y]}Z$
\end{definition}

Suppose $F:M \to N$ is a Riemannian isometry

\begin{claim}
    We have $\nabla_{F_*X}(F_*Y)=F_*(\nabla_XY)$
\end{claim}

\begin{claim}
    We have $\nabla_{F_*v} \circ F_* = DF_p \circ \nabla_v$
\end{claim}

\begin{claim}
    We have $F_*[R(X,Y,Z)]=R(F_*X,F_*Y,F_*Z)$
\end{claim}

\section{Vector fields along a curve}

\begin{definition}
    Suppose $\gamma: I \to M$ is a smooth curve. A smooth vector field along $\gamma$ is a smooth map $V: I \to \mathbb{R}^k$ such that $V(t) \in T_{\gamma (t)}M$
\end{definition}

The vector space of smooth vector fields along $\gamma$ is $\mathcal{X}(\gamma)$

\begin{definition}
    We have  $\text{cov} \in L(\mathcal{X}(\gamma))$ given by $V \to (t \to \pi_{\gamma(t)}\dot{V}(t))$
\end{definition}

\begin{definition}
    A vector field $V$ along $\gamma$ is called \textbf{parallel} to $\gamma$ if $V \in \text{ker} (\text{cov}_\gamma)$
\end{definition}

\begin{claim}
    Suppose $t_0 \in I$, $p_0=\gamma(t_0)$ and $v_0 \in T_{p_0}M$. There is a unique parallel vector field $V$ satisfying $V(t_0)=v_0$
\end{claim}

\begin{definition}
    Let $t_1,t_2 \in I$. We have the \textbf{parallel transport} map $P_{t_1t_2} \in L(T_{\gamma(t_1)}M,T_{\gamma(t_2)}M)$ 
\end{definition}

\begin{claim}
    $P_{t_1t_2}$ is an isometry
\end{claim}

\begin{claim}
    $P_{t_1t_2}$ is invertible with inverse $P_{t_2t_1}$
\end{claim}

\begin{claim}
    Suppose $t_0$ is in the interior of $I$. Then we have \[\nabla_vX= \text{lim}_{t \to t_0}\frac{P_{tt_0}(X(\gamma(t)))- X_p}{t-t_0}\]
\end{claim}

\section{Geodesics}

\begin{definition}
    A smooth curve $\gamma$ is a \textbf{geodesic} if $\gamma'$ is parallel to $\gamma$
\end{definition}

\begin{claim}
  If $\gamma$ is a geodesic then $||\gamma'(t_1)||=||\gamma'(t_2)||$ for all $t_1,t_2$
\end{claim}

Suppose $p \in M$ and $v \in T_pM$. Consider geodesics $\gamma$ such that $\gamma(0)=p$ and $\gamma'(0)=v$

\begin{claim}
    There is such a geodesic $\gamma$ on $(-\epsilon,\epsilon)$
\end{claim}

\begin{claim}
    Any two such geodesics agree on their common domain
\end{claim}

\begin{claim}
    There is a unique \textbf{maximal geodesic} $\gamma_v$ of this kind on an open interval $I_v$
\end{claim}

\begin{claim}
    $I_{cv}=\frac{I_v}{c}$ and $\gamma_{cv}(t)=\gamma_v(ct)$
\end{claim}

\subsection{Relation with length minimizing cuves}

\begin{claim}
    If $\gamma$ is a geodesic then for each $t$ we have that $\gamma_{[a,b]}$ is length minimizing where $t \in [a,b]$
\end{claim}

\begin{claim}
    Suppose $\gamma$ is piecewise regular and length minimizing. Then $\gamma$ is smooth 
\end{claim}

\begin{claim}
    Suppose $\gamma$ is regular and length minimizing. Then the arc length parametrization of $\gamma$ is a geodesic
\end{claim}

\section{Exponential map}

\begin{definition}
    We define $E_p =\{v \in T_pM:1 \in I_v\}$
\end{definition}

\begin{claim}
    $E_p$ is star convex with respect to $0$
\end{claim}

\begin{claim}
    $E_p$ is open in $T_pM$
\end{claim}

\begin{claim}
    $I_v=\{t:tv \in E_p\}$
\end{claim}

\begin{definition}
    We define $\exp_p \in C^\infty(E_p,M)$ by $v \to \gamma_v(1)$
\end{definition}

\begin{claim}
    $(D\exp_p)_0=I$
\end{claim}

\begin{claim}
    $\exp_p$ is a local diffeomorphism at $0$
\end{claim}

\begin{claim}
    $\gamma_v(t)=\exp(tv)$
\end{claim} 

\begin{definition}
    We define $E \subseteq TM$ as $\{(p,v): v \in E_p\}$
\end{definition}

\begin{claim}
    $E$ is open in $TM$
\end{claim}

\begin{definition}
    We define $\exp \in C^\infty(E,M)$ by $(p,v) \to \exp_p(v)$
\end{definition}

Suppose $F:M \to N$ is a local Riemannian isometry at $p$

\begin{claim}
  For all $v$ in $T_pM$ we have $F_*(v) \in \mathcal{E}_{F(p)}$
\end{claim}

\begin{claim}
  We have $\exp_{F(p)}(F_*(v))=F(\exp_p(v))$
\end{claim}

\section{Normal coordinate systems}

\begin{definition}
  A neighborhood $U \subseteq E_p$ of the origin is called \textbf{prenormal} if $\exp_p:U \to \exp(U)$ is a diffeomorphism.
\end{definition}

\begin{definition}
  A \textbf{normal neighborhood} is the image under $\exp_p$ of a star convex prenormal neighborhood
\end{definition}

\begin{definition}
  Suppose $v_1,v_2,\dots,v_n$ is an orthonormal basis of $T_pM$ and $U$ is a normal neighborhood of $p$ with corresponding prenormal neighborhood $V$. 
  Define $\Omega=\{x \in \mathbb{R}^n : \sum_{i=1}^n x_iv_i \in V\}$ and $\phi \in \text{Diffeo}(\Omega,U)$ by $\phi(x)=\exp_p(\sum x_iv_i)$
  Then $\phi$ is called a \textbf{normal coordinate system} centered at $p$
\end{definition}

\begin{claim}
  If $\phi$ is a normal coordinate system on $U$ then $\phi_i(p)$ is an orthonormal basis of $T_pM$
\end{claim}

\begin{claim}
  If $\phi \in \text{Diffeo}(\Omega_1,U)$ and $\psi \in \text{Diffeo}(\Omega_2,U)$ are normal coordinate systems then there is $A \in O_n$ such that 
  \begin{itemize}
    \item $\Omega_2=A(\Omega_1)$
    \item $\psi \circ A \circ \phi^{-1}=\text{id}_U$
  \end{itemize}
  
\end{claim}

\begin{claim}
  Suppose $\phi \in \text{Diffeo}(\Omega,U)$ is a normal coordinate system centered at $p$ with $\phi_i(p)=v_i$. Then the following hold:
  \begin{itemize}
    \item $\Gamma_{ij}^k(0)=0$
    \item $G_{ij}(p)=\delta_{ij}$
    \item Whenever $tx \in \Omega$ we have $\gamma_v(t)=\phi(tx)$ where $v=\sum x_iv_i$
    \item $(DG_{ij})_p(D\phi)_0=0$
    \item For $q =\exp_p(v) \in U$ we have 
      \begin{itemize}
        \item $r(q)=||v||$
        \item $||v||\phi_r(q)=\gamma_v'(1)$
        \item $||\phi_r(q)||=1$
      \end{itemize}
    \item $\text{grad }r=\phi_r$
  \end{itemize}
\end{claim}


\section{Injectivity radius}

\begin{definition}
  The \textbf{injectivity radius} is defined as \[\text{inj}_p=\text{sup }\{r > 0: B_r \text{ is prenormal}\}\]
\end{definition}

\begin{claim}
  If $\rho=\text{inj}_p$, then $B_\rho$ is prenormal
\end{claim}

\begin{claim}
  Suppose $0<\epsilon<\rho$. Then $\Sigma_\epsilon=\exp_p(\delta B_\epsilon)$ is a submanifold of $M$
\end{claim}

\begin{claim}
  Suppose $\phi \in \text{Diffeo}(\Omega,U)$ is a normal coordinate system centered at $p$. Let $q =\exp_p(v) \in U$ with $\epsilon= ||v||< \rho$. Then $\phi_r(q) \in T_q\Sigma_\epsilon$  
\end{claim}

\begin{claim}
  Suppose $q=\exp_p(v)$ where $||v||<\rho$. Consider the curve $\gamma:[0,1] \to M$ given by $\gamma(t)=\exp_p(tv)$. Then we have the following:
  \begin{itemize}
    \item $\gamma$ is minimizing
    \item Any other minimizing curve from $p$ to $q$ is a reparametrization of $\gamma$
  \end{itemize}
\end{claim}

\begin{definition}
  A neighborhood $V$ of $p$ is $\delta$- uniformly normal if for all $q \in V$ we have
  \begin{itemize}
    \item $\delta < \text{inj}_q$
    \item $V \subseteq \exp_q(B_\delta)$
  \end{itemize}
\end{definition}

\begin{definition}
  A neighborhood is uniformly normal if it is $\delta$-uniformly normal for some $\delta$
\end{definition}

\begin{claim}
  There is a uniformly normal neighborhood around every point
\end{claim}

\section{One parameter family of curves}

\begin{definition}
We call $\Gamma \in C^\infty(I \times J,M)$ a \textbf{one parameter family of curves} on $J$
\end{definition}

\begin{definition}
  The \textbf{variation field} $V \in \mathcal{X}(\Gamma_0)$ of $\Gamma$ is defined by $t \to \partial_1\Gamma(0,t)$
\end{definition}

\begin{claim}
  We have $[\text{cov}(\partial_1\Gamma)_s](t)=[\text{cov}(\partial_2\Gamma)_t](s)$
\end{claim}

Suppose $\gamma:[a,b] \to M$ denotes a regular curve from $p$ to $q$

\begin{definition}
	A \textbf{proper variation} of $\gamma$ is a map $\Gamma \in 
	C^{\infty}((-\epsilon,\epsilon) \times K,M)$ such that 
	$\Gamma_{0}=\gamma$ and $\Gamma_{s}$ is a regular curve from $p$ to $q$
\end{definition}

\begin{claim}
  Suppose $V \in \mathcal{X}(\gamma)$ with $V(a)=V(b)=0$. Then there is a proper variation $\Gamma$ of $\gamma$ whose variation field is $V$
\end{claim}

\section{Second fundamental form}

\begin{definition}
    $N_pM$ is the orthogonal complement of $T_pM$ in $\mathbb{R}^k$
\end{definition}

\begin{definition}
    The \textbf{second fundamental form} of $M$ at $p$ is the bilinear map $S_p: T_pM \times T_pM \to \mathbb{R}^k$ given by $v,w \to [(D\pi)_pv]w$
\end{definition}

\begin{claim}
    $\text{range }S_p \subseteq N_pM$ for all $p$
\end{claim}

\begin{claim}
    The second fundamental form is symmetric
\end{claim}

\begin{claim}
    We have $S_p(X_p,Y_p)=\hat{\pi}_p((D_XY)(p))$
\end{claim}
